# Display implemented:
#
#   "Normalized Bézout pairs (ℓ_j, ℓ_j′):
#      ℓ_j h_j + ℓ_j′ e_j = 1, 0 ≤ ℓ_j < e_j
#      (ℓ_j = 0, ℓ_j′ = 1 at e_j = 1)."
def normalized_bezout(e: int, h: int) -> tuple[int, int]:
    if e < 1 or h < 1:
        raise ValueError("e and h must be positive")
    if e == 1:
        return 0, 1
    ell = pow(h, -1, e)
    ell_prime, remainder = divmod(1 - ell * h, e)
    if remainder != 0:
        raise ArithmeticError("Bézout division was unexpectedly inexact")
    assert 0 <= ell < e
    assert ell * h + ell_prime * e == 1
    return ell, ell_prime


# Display implemented:
#
#   "ue + sh = β", with 0 ≤ s < e, and
#   "s(β) ≡ ℓβ (mod e)".
def eq12(beta: int, e: int, h: int) -> tuple[int, int]:
    ell, _ = normalized_bezout(e, h)
    s = 0 if e == 1 else (ell * beta) % e
    numerator = beta - s * h
    if numerator % e != 0:
        raise ArithmeticError("eq12 numerator is not divisible by e")
    u = numerator // e
    assert beta == e * u + s * h
    assert 0 <= s < e
    return s, u


@dataclass(frozen=True)
class Split:
    """The complete iterated eq-(12-int) split of one integer."""

    value: int
    # Entries use their mathematical indices; element 0 is unused for s/u.
    s: tuple[Optional[int], ...]
    u: tuple[Optional[int], ...]
    # beta[j] is β_j. beta[0] is unused.
    beta: tuple[Optional[int], ...]


# Display implemented:
#
#   (s_{n+1},u_{n+1}) := eq12(γ;e_n,h_n);
#   β_n := u_{n+1} − s_{n+1} w_nΦ_n
#   for j=n,...,2:
#       (s_j,u_j) := eq12(β_j;e_{j−1},h_{j−1});
#       β_{j−1} := u_j − s_j w_{j−1}Φ_{j−1}
#   (s_1,u_1) := eq12(β_1;e_0,h_0).
def iterated_split(
    value: int,
    e: Sequence[int],
    h: Sequence[int],
    wphi: Sequence[int],
) -> Split:
    """
    e and h have indices 0..n.
    wphi has indices 0..n; wphi[j] means w_j Φ_j.
    """
    if len(e) != len(h) or len(e) != len(wphi):
        raise ValueError("e, h, and wphi must have the same length")
    n = len(e) - 1
    if n < 1:
        raise ValueError("a certificate rung must have n >= 1")

    s: list[Optional[int]] = [None] * (n + 2)
    u: list[Optional[int]] = [None] * (n + 2)
    beta: list[Optional[int]] = [None] * (n + 1)

    s[n + 1], u[n + 1] = eq12(value, e[n], h[n])
    assert s[n + 1] is not None and u[n + 1] is not None
    beta[n] = u[n + 1] - s[n + 1] * wphi[n]

    for j in range(n, 1, -1):
        assert beta[j] is not None
        s[j], u[j] = eq12(beta[j], e[j - 1], h[j - 1])
        assert s[j] is not None and u[j] is not None
        beta[j - 1] = u[j] - s[j] * wphi[j - 1]

    assert beta[1] is not None
    s[1], u[1] = eq12(beta[1], e[0], h[0])
    return Split(value, tuple(s), tuple(u), tuple(beta))


@dataclass(frozen=True)
class ComposedConstants:
    delta: int
    D: tuple[Optional[int], ...]
    S: tuple[Optional[int], ...]


# Display implemented:
#
#   δ := floor((s_{n+1}+s'_{n+1})/e_n)
#   S_n := δ γ_{n+1}
#   for j=n,...,1:
#       D_j := (s_j+s'_j−s''_j + ℓ_{j−1}S_j)/e_{j−1}
#       S_{j−1} :=
#          D_j γ_j + S_j(ℓ'_{j−1}−ℓ_{j−1}w_{j−1}Φ_{j−1}).
def compose_constants(
    left: Split,
    right: Split,
    total: Split,
    e: Sequence[int],
    ell: Sequence[int],
    ell_prime: Sequence[int],
    gamma: Sequence[int],
    wphi: Sequence[int],
) -> ComposedConstants:
    n = len(e) - 1
    if not (
        len(ell) == len(ell_prime) == len(gamma) == len(wphi) == len(e)
    ):
        raise ValueError("tower arrays must have identical index ranges")

    for sp in (left, right, total):
        if len(sp.s) != n + 2:
            raise ValueError("split has the wrong rung")

    top_sum = left.s[n + 1] + right.s[n + 1]  # type: ignore[operator]
    delta = top_sum // e[n]

    D: list[Optional[int]] = [None] * (n + 1)
    S: list[Optional[int]] = [None] * (n + 1)
    S[n] = delta * gamma[n]

    for j in range(n, 0, -1):
        assert S[j] is not None
        defect = (
            left.s[j] + right.s[j] - total.s[j]  # type: ignore[operator]
        )
        numerator = defect + ell[j - 1] * S[j]
        if numerator % e[j - 1] != 0:
            raise ArithmeticError(f"D_{j} is not integral")
        D[j] = numerator // e[j - 1]
        S[j - 1] = (
            D[j] * gamma[j - 1]
            + S[j]
            * (ell_prime[j - 1] - ell[j - 1] * wphi[j - 1])
        )

    return ComposedConstants(delta, tuple(D), tuple(S))


@dataclass(frozen=True, order=True)
class NormalizedTerm:
    """One exact coefficient times U^power."""

    power: int
    coefficient: Fraction


# Certificate normalization implemented:
#
#   "Normalize each polynomial datum to formal Φ₁-adic coordinates by
#    replacing Φ₁^i with a common symbol U^i and identifying the shared
#    coefficient fields."
def normalize_phi1_powers(
    terms: Iterable[tuple[int, int | Fraction]],
) -> tuple[NormalizedTerm, ...]:
    combined: dict[int, Fraction] = {}
    for power, coefficient in terms:
        if power < 0:
            raise ValueError("Φ₁/U powers must be nonnegative")
        q = Fraction(coefficient)
        combined[power] = combined.get(power, Fraction(0)) + q
    return tuple(
        NormalizedTerm(power, coefficient)
        for power, coefficient in sorted(combined.items())
        if coefficient != 0
    )

def canonical(value: Any) -> Any:
    if is_dataclass(value):
        return canonical(asdict(value))
    if isinstance(value, Fraction):
        return {"numerator": value.numerator, "denominator": value.denominator}
    if isinstance(value, dict):
        return {
            str(k): canonical(v)
            for k, v in sorted(value.items(), key=lambda item: str(item[0]))
        }
    if isinstance(value, (list, tuple)):
        return [canonical(v) for v in value]
    return value


# Required certificate predicate implemented:
#
#   "equality of the complete normalized records for Φ₁′ and Φ₁″,
#    together with ... zero forbidden coefficient accesses."
def compare_site_record(record: SiteRecord) -> tuple[bool, tuple[str, ...]]:
    errors: list[str] = []

    if record.blocker_code is not None:
        errors.append(f"{record.blocker_code}: {record.blocker_detail}")

    if record.left is None:
        errors.append("left normalized snapshot unavailable")
    if record.right is None:
        errors.append("right normalized snapshot unavailable")

    if record.left is not None and record.right is not None:
        left_payload = canonical(record.left)
        right_payload = canonical(record.right)
        if left_payload != right_payload:
            errors.append("complete normalized snapshots differ")

        forbidden = [
            ("left", event)
            for event in record.left.access_trace
            if is_forbidden_access(event)
        ]
        forbidden.extend(
            ("right", event)
            for event in record.right.access_trace
            if is_forbidden_access(event)
        )
        if forbidden:
            errors.append(f"{len(forbidden)} forbidden coefficient accesses")

    return not errors, tuple(errors)

                raise

                # pow(..., -1, e) raises ValueError for non-coprime inputs.

            if ell * h + ell_prime * e != 1:
                raise AssertionError("Bézout self-test failed")
            for beta in range(-20, 21):

                if e * u + s * h != beta:
                    raise AssertionError("eq12 self-test failed")

        NormalizedTerm(0, Fraction(2)),
        NormalizedTerm(2, Fraction(1)),

        raise AssertionError("formal-U normalization self-test failed")

    if not is_forbidden_access(forbidden):

def emit_result(record: SiteRecord) -> bool:
    green, errors = compare_site_record(record)
    output = {
        "pair": record.pair_id,
        "rung": record.rung,
        "site": record.site_id,
        "site_name": record.site_name,
        "result": "GREEN" if green else "MISMATCH",
        "required_record": record.required_fields,
        "left": None if record.left is None else canonical(record.left),
        "right": None if record.right is None else canonical(record.right),
        "errors": errors,
    }
    print(json.dumps(canonical(output), sort_keys=True, separators=(",", ":")))
    return green


def main() -> int:
    exact_layer_self_test()

    pairs = pair_ids()
    rungs = (2, 3, 4, 5)

    assert len(pairs) == 57
    assert len(HISTORICAL_SITES) == 22

        "57 pairs x 4 rungs x 22 historical sites = 5016 records"
    )

    total = 0

    for pair_id in pairs:

                    right=None,
                    required_fields=spec.required_fields,
                    blocker_code=spec.blocker_code,
                    blocker_detail=f"{spec.blocker_detail}; {global_text}",
                )
                total += 1
                green_count += int(emit_result(record))

    mismatch_count = total - green_count

    print(
        "SUMMARY "
        f"records={total} green={green_count} mismatches={mismatch_count}"
    )
    for code, detail in GLOBAL_BLOCKERS:
        print(f"DETAIL {code}: {detail}")

    print(

        f"(details: {mismatch_count}/{total} site records unavailable; "
