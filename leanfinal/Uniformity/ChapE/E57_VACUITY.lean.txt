/-- **The committed E.57 signature, byte-for-byte** (`leanspec/Leanspec/ChapE.lean:1159-1171`,
blueprint §8 NODE E.57), proved outright. All three hypotheses — both carrier legs and the
`hpart : True` GC-13 placeholder — are inert; the Lean linter flags all three as unreferenced. -/
theorem block_split_IS_A_THEOREM {O : Type*} [CommRing O] [IsDomain O] {K : Type*} [Field K]
    {C : SlotCarrier O K} {B : BlockData C} (I : RungInterface C B)
    (hblocks : ∀ p ∈ I.sides, ∀ q ∈ I.linFac p,
      ∃ Fpq : Polynomial O, Fpq.Monic ∧ Fpq ∣ B.F ∧
        Fpq.natDegree = I.classCount p q)
    (hblocksHi : ∀ p ∈ I.sides, ∀ q ∈ I.hiFac p,
      ∃ Fpq : Polynomial O, Fpq.Monic ∧ Fpq ∣ B.F ∧
        Fpq.natDegree = I.classCountHi p q)
    (hpart : True) :
    ∃ blocks : List (Polynomial O),
        B.F = blocks.prod ∧
        (blocks.map Polynomial.natDegree).sum = B.F.natDegree :=
  ⟨[B.F], by simp, by simp⟩

/-- The sharper form: the conclusion needs no rung interface, no block data, no domain
hypothesis and no carrier — just a polynomial. This is why typing `hpart` cannot rescue the
node: the strength was never in the displayed conclusion. -/
theorem block_split_conclusion_from_nothing {O : Type*} [CommRing O] (F : Polynomial O) :
    ∃ blocks : List (Polynomial O),
        F = blocks.prod ∧
        (blocks.map Polynomial.natDegree).sum = F.natDegree :=
  ⟨[F], by simp, by simp⟩

/-- And it survives an arbitrary typing of the socket: for every `P`, the committed shape with
`hpart : P` is provable. The quantifier over `P` makes the orthogonality explicit. -/
theorem socket_typing_is_orthogonal {O : Type*} [CommRing O] (P : Prop) (F : Polynomial O) :
    P → ∃ blocks : List (Polynomial O),
        F = blocks.prod ∧
        (blocks.map Polynomial.natDegree).sum = F.natDegree :=
  fun _ => block_split_conclusion_from_nothing F
