theorem addVal_two_pow (h2 : Irreducible (2 : O)) (n : ℕ) :

theorem isUnit_three (h2 : Irreducible (2 : O)) : IsUnit (3 : O) := by
  by_contra hc
  have hmem : (3 : O) ∈ IsLocalRing.maximalIdeal O := hc
  have h2mem : (2 : O) ∈ IsLocalRing.maximalIdeal O := h2.not_isUnit
  have h1 : (1 : O) ∈ IsLocalRing.maximalIdeal O := by
    have : (3 : O) - 2 = 1 := by norm_num
    rw [← this]
    exact sub_mem hmem h2mem
  exact (IsLocalRing.maximalIdeal.isMaximal O).ne_top (Ideal.eq_top_of_isUnit_mem _ h1 isUnit_one)

theorem addVal_twelve (h2 : Irreducible (2 : O)) : addVal O (12 : O) = 2 := by

theorem fw_expansion : (fw : Polynomial O) = C 32 + C 12 * s2Key O + s2Key O ^ 2 := by
  have h84 : (C 8 : Polynomial O) * C 4 = C 32 := by rw [← C_mul]; norm_num
  have h8p4 : (C 8 : Polynomial O) + C 4 = C 12 := by rw [← C_add]; norm_num
  rw [fw, gsix, gfour]
  calc (s2Key O + C 8) * (s2Key O + C 4)
      = s2Key O ^ 2 + (C 8 + C 4) * s2Key O + C 8 * C 4 := by ring
    _ = C 32 + C 12 * s2Key O + s2Key O ^ 2 := by rw [h84, h8p4]; ring

theorem dev_fw : dev (s2Key O) (fw : Polynomial O) 0 = C 32
    ∧ dev (s2Key O) (fw : Polynomial O) 1 = C 12
    ∧ dev (s2Key O) (fw : Polynomial O) 2 = 1 := by
  classical
  have hdegK : (0 : ℕ) < (s2Key O).natDegree := by rw [s2Key_natDegree]; norm_num
  have hdegC : ∀ c : O, (C c : Polynomial O).degree < (s2Key O).degree := by
    intro c
    refine lt_of_le_of_lt (degree_C_le) ?_
    rw [Polynomial.degree_eq_natDegree s2Key_monic.ne_zero, s2Key_natDegree]
    exact_mod_cast Nat.zero_lt_two
  set A : Fin 2 → Polynomial O := ![C 32, C 12] with hA
  have hAdeg : ∀ j : Fin 2, (A j).degree < (s2Key O).degree := by
    intro j
    fin_cases j <;> simp [hA] <;> exact hdegC _
  have hsum : fw = ∑ i : Fin 2, A i * s2Key O ^ (i : ℕ) + s2Key O ^ 2 := by
    rw [fw_expansion, Fin.sum_univ_two]
    simp only [hA, Matrix.cons_val_zero, Matrix.cons_val_one, Matrix.head_cons,
      Fin.val_zero, Fin.val_one, pow_zero, pow_one, mul_one]
  obtain ⟨hlow, htop⟩ := C109asm.dev_assemble s2Key_monic hdegK A hAdeg
  refine ⟨?_, ?_, ?_⟩
  · have := hlow 0
    rw [← hsum] at this
    simpa [hA] using this
  · have := hlow 1
    rw [← hsum] at this
    simpa [hA] using this
  · rw [hsum]
    exact htop

/-! ### 2c. Side-set arithmetic — membership unpacked, the witness side data in ℕ -/

/-- C.07 membership, unpacked as an iff (C.08's private filter plumbing, both directions). -/
theorem mem_dvSideSet_iff {π : O} {F : KeyFrame O π} {f : Polynomial O} {u ℓ j : ℕ} :
    j ∈ dvSideSet F f u ℓ ↔
      j ≤ f.natDegree ∧ dvSupp F f u ℓ = ℓ • dvHgt F f j + (u * j : ℕ∞) ∧
        dvHgt F f j ≠ ⊤ := by
  classical
  constructor
  · intro hj
    have hj' : j ∈ Finset.filter (DvOnSide F f u ℓ) (Finset.range (f.natDegree + 1)) := hj
    obtain ⟨hr, hs1, hs2⟩ := Finset.mem_filter.mp hj'
    exact ⟨Nat.lt_succ_iff.mp (Finset.mem_range.mp hr), hs1, hs2⟩
  · rintro ⟨hr, hs1, hs2⟩
    show j ∈ Finset.filter (DvOnSide F f u ℓ) (Finset.range (f.natDegree + 1))
    exact Finset.mem_filter.mpr ⟨Finset.mem_range.mpr (Nat.lt_succ_iff.mpr hr), ⟨hs1, hs2⟩⟩

/-- a finite `dv`-term collapses to a single cast. -/
theorem term_cast {x : ℕ∞} {a : ℕ} (hx : x = (a : ℕ∞)) (ℓ u b : ℕ) :
    ℓ • x + ((u : ℕ∞) * (b : ℕ∞)) = ((ℓ * a + u * b : ℕ) : ℕ∞) := by
  subst hx
  rw [nsmul_eq_mul]
  push_cast
  ring

/-- the cleared support of `fw`, in ℕ: the min of the three finite terms
`{ℓ·10, ℓ·4 + u, u·2}` (heights `[10, 4, 0]`, abscissae `0, 1, 2`; the `⊤` tail drops). -/
theorem dvSupp_fw {u ℓ : ℕ} (hl : 0 < ℓ) :
    dvSupp (s2Frame h2 hq) (fw : Polynomial O) u ℓ
      = ((min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) : ℕ) : ℕ∞) := by

  have t1 := term_cast hd1 ℓ u 1
  have t2 := term_cast hd2 ℓ u 2
  rw [show dvSupp (s2Frame h2 hq) (fw : Polynomial O) u ℓ
      = (Finset.range ((fw : Polynomial O).natDegree + 1)).inf
          (fun j => ℓ • dvHgt (s2Frame h2 hq) (fw : Polynomial O) j + (u * j : ℕ∞))
      from rfl, natDegree_fw]
  apply le_antisymm
  · have hchoice : (min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2)
          = ℓ * 10 + u * 0)
        ∨ (min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) = ℓ * 4 + u * 1)
        ∨ (min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2)
          = ℓ * 0 + u * 2) := by omega
    rcases hchoice with h | h | h <;> rw [h]
    · rw [← t0]
      exact Finset.inf_le (Finset.mem_range.mpr (by norm_num))
    · rw [← t1]
      exact Finset.inf_le (Finset.mem_range.mpr (by norm_num))
    · rw [← t2]
      exact Finset.inf_le (Finset.mem_range.mpr (by norm_num))
  · refine Finset.le_inf fun j hj => ?_
    have hj4 : j ≤ 4 := Nat.lt_succ_iff.mp (Finset.mem_range.mp hj)
    interval_cases j
    · rw [t0]
      exact_mod_cast (by omega :
        min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) ≤ ℓ * 10 + u * 0)
    · rw [t1]
      exact_mod_cast (by omega :
        min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) ≤ ℓ * 4 + u * 1)
    · rw [t2]
      exact_mod_cast (by omega :
        min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) ≤ ℓ * 0 + u * 2)
    · rw [hdtop 3 (by norm_num), C118a.smul_top_pos hl, top_add]

    · rw [hdtop 4 (by norm_num), C118a.smul_top_pos hl, top_add]

    j ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ ↔
      (j = 0 ∧ min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) = ℓ * 10 + u * 0)
      ∨ (j = 1 ∧ min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2) = ℓ * 4 + u * 1)
      ∨ (j = 2 ∧ min (min (ℓ * 10 + u * 0) (ℓ * 4 + u * 1)) (ℓ * 0 + u * 2)
          = ℓ * 0 + u * 2) := by
  obtain ⟨hd0, hd1, hd2, hdtop⟩ := dvHgt_fw h2 hq
  have t0 := term_cast hd0 ℓ u 0
  have t1 := term_cast hd1 ℓ u 1
  have t2 := term_cast hd2 ℓ u 2
  rw [mem_dvSideSet_iff, natDegree_fw, dvSupp_fw h2 hq hl]
  constructor
  · rintro ⟨hj4, heq, htne⟩
    interval_cases j
    · refine Or.inl ⟨rfl, ?_⟩
      rw [t0] at heq
      exact_mod_cast heq
    · refine Or.inr (Or.inl ⟨rfl, ?_⟩)
      rw [t1] at heq
      exact_mod_cast heq
    · refine Or.inr (Or.inr ⟨rfl, ?_⟩)
      rw [t2] at heq
      exact_mod_cast heq
    · exact absurd (hdtop 3 (by norm_num)) htne
    · exact absurd (hdtop 4 (by norm_num)) htne
  · rintro (⟨rfl, hm⟩ | ⟨rfl, hm⟩ | ⟨rfl, hm⟩)
    · exact ⟨by norm_num, by rw [t0]; exact_mod_cast hm, by rw [hd0]; simp⟩
    · exact ⟨by norm_num, by rw [t1]; exact_mod_cast hm, by rw [hd1]; simp⟩
    · exact ⟨by norm_num, by rw [t2]; exact_mod_cast hm, by rw [hd2]; simp⟩

/-- the cleared support of a factor `Φ′ + 2ⁿ`, in ℕ: the min of the two finite terms
`{ℓ·2n, u}` (heights `[2n, 0]`, abscissae `0, 1`; the height at `2` is `⊤`). -/
theorem dvSupp_g {n u ℓ : ℕ} (hn : 0 < n) (hl : 0 < ℓ) :
    dvSupp (s2Frame h2 hq) (s2Key O + C (2 ^ n : O)) u ℓ
      = ((min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) : ℕ) : ℕ∞) := by
  obtain ⟨hd0, hd1, hdtop⟩ := dvHgt_g h2 hq hn
  have t0 := term_cast hd0 ℓ u 0
  have t1 := term_cast hd1 ℓ u 1
  rw [show dvSupp (s2Frame h2 hq) (s2Key O + C (2 ^ n : O)) u ℓ
      = (Finset.range ((s2Key O + C (2 ^ n : O)).natDegree + 1)).inf
          (fun j => ℓ • dvHgt (s2Frame h2 hq) (s2Key O + C (2 ^ n : O)) j + (u * j : ℕ∞))
      from rfl, natDegree_key_add_C]
  apply le_antisymm
  · have hchoice : (min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) = ℓ * (2 * n) + u * 0)
        ∨ (min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) = ℓ * 0 + u * 1) := by omega
    rcases hchoice with h | h <;> rw [h]
    · rw [← t0]
      exact Finset.inf_le (Finset.mem_range.mpr (by norm_num))
    · rw [← t1]
      exact Finset.inf_le (Finset.mem_range.mpr (by norm_num))
  · refine Finset.le_inf fun j hj => ?_
    have hj2 : j ≤ 2 := Nat.lt_succ_iff.mp (Finset.mem_range.mp hj)
    interval_cases j
    · rw [t0]
      exact_mod_cast (by omega :
        min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) ≤ ℓ * (2 * n) + u * 0)
    · rw [t1]
      exact_mod_cast (by omega :
        min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) ≤ ℓ * 0 + u * 1)
    · rw [hdtop 2 (by norm_num), C118a.smul_top_pos hl, top_add]
      exact le_top

/-- membership in the side set of a factor `Φ′ + 2ⁿ`, fully in ℕ. -/
theorem mem_dvSideSet_g {n u ℓ j : ℕ} (hn : 0 < n) (hl : 0 < ℓ) :
    j ∈ dvSideSet (s2Frame h2 hq) (s2Key O + C (2 ^ n : O)) u ℓ ↔
      (j = 0 ∧ min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) = ℓ * (2 * n) + u * 0)
      ∨ (j = 1 ∧ min (ℓ * (2 * n) + u * 0) (ℓ * 0 + u * 1) = ℓ * 0 + u * 1) := by
  obtain ⟨hd0, hd1, hdtop⟩ := dvHgt_g h2 hq hn
  have t0 := term_cast hd0 ℓ u 0
  have t1 := term_cast hd1 ℓ u 1
  rw [mem_dvSideSet_iff, natDegree_key_add_C, dvSupp_g h2 hq hn hl]
  constructor
  · rintro ⟨hj2, heq, htne⟩
    interval_cases j
    · refine Or.inl ⟨rfl, ?_⟩
      rw [t0] at heq
      exact_mod_cast heq
    · refine Or.inr ⟨rfl, ?_⟩
      rw [t1] at heq
      exact_mod_cast heq
    · exact absurd (hdtop 2 (by norm_num)) htne
  · rintro (⟨rfl, hm⟩ | ⟨rfl, hm⟩)
    · exact ⟨by norm_num, by rw [t0]; exact_mod_cast hm,
        by rw [hd0]; exact ENat.coe_ne_top _⟩
    · exact ⟨by norm_num, by rw [t1]; exact_mod_cast hm, by rw [hd1]; simp⟩

/-- `dvSideDeg` of `fw` at `(6,1)` is `1` (side set `{0, 1}`). -/
theorem dvSideDeg_fw_six
    (hne : (dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 6 1).Nonempty) :
    dvSideDeg (s2Frame h2 hq) (fw : Polynomial O) 6 1 hne = 1 := by
  have h0 : 0 ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 6 1 := by
    rw [mem_dvSideSet_fw h2 hq Nat.one_pos]
    exact Or.inl ⟨rfl, by omega⟩
  have h1 : 1 ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 6 1 := by
    rw [mem_dvSideSet_fw h2 hq Nat.one_pos]
    exact Or.inr (Or.inl ⟨rfl, by omega⟩)
  have hub : ∀ x ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 6 1, x ≤ 1 := by
    intro x hx
    rw [mem_dvSideSet_fw h2 hq Nat.one_pos] at hx
    rcases hx with ⟨rfl, -⟩ | ⟨rfl, -⟩ | ⟨rfl, hx2⟩ <;> omega
  have hmin : dvSideMin (s2Frame h2 hq) (fw : Polynomial O) 6 1 hne = 0 :=
    Nat.le_antisymm (Finset.min'_le _ 0 h0) (Nat.zero_le _)
  have hmax : dvSideMax (s2Frame h2 hq) (fw : Polynomial O) 6 1 hne = 1 :=
    Nat.le_antisymm (Finset.max'_le _ hne _ hub) (Finset.le_max' _ 1 h1)
  show (dvSideMax (s2Frame h2 hq) (fw : Polynomial O) 6 1 hne
      - dvSideMin (s2Frame h2 hq) (fw : Polynomial O) 6 1 hne) / 1 = 1
  rw [hmax, hmin]

/-- `dvSideDeg` of `fw` at `(4,1)` is `1` (side set `{1, 2}`). -/
theorem dvSideDeg_fw_four

  have hlb : ∀ x ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 4 1, 1 ≤ x := by

  have hub : ∀ x ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 4 1, x ≤ 2 := by
    intro x hx
    rw [mem_dvSideSet_fw h2 hq Nat.one_pos] at hx

  have hmin : dvSideMin (s2Frame h2 hq) (fw : Polynomial O) 4 1 hne = 1 :=

  slopes := wSlopes
  factor := wFactor
  below := 1
  hslopes := by
    intro p hp
    have hp' : p = ((6 : ℕ), (1 : ℕ)) ∨ p = ((4 : ℕ), (1 : ℕ)) := by
      simpa [wSlopes, Prod.ext_iff] using hp
    obtain ⟨he, hf, hh, -⟩ := s2Frame_data h2 hq
    rw [he, hf, hh]
    rcases hp' with rfl | rfl <;>
      exact ⟨Nat.one_pos, Nat.coprime_one_right _, by norm_num⟩
  hdistinct := by
    intro p hp q hq' hpq
    have hp' : p = ((6 : ℕ), (1 : ℕ)) ∨ p = ((4 : ℕ), (1 : ℕ)) := by
      simpa [wSlopes, Prod.ext_iff] using hp
    have hq'' : q = ((6 : ℕ), (1 : ℕ)) ∨ q = ((4 : ℕ), (1 : ℕ)) := by
      simpa [wSlopes, Prod.ext_iff] using hq'
    rcases hp' with rfl | rfl <;> rcases hq'' with rfl | rfl <;>
      first
        | exact absurd rfl hpq
        | norm_num
  hmonic := by
    intro p hp
    have hp' : p = ((6 : ℕ), (1 : ℕ)) ∨ p = ((4 : ℕ), (1 : ℕ)) := by
      simpa [wSlopes, Prod.ext_iff] using hp
    rcases hp' with rfl | rfl
    · rw [wFactor_six]; exact monic_gsix
    · rw [wFactor_four]; exact monic_gfour
  hpure := by
    intro p hp
    have hp' : p = ((6 : ℕ), (1 : ℕ)) ∨ p = ((4 : ℕ), (1 : ℕ)) := by
      simpa [wSlopes, Prod.ext_iff] using hp
    obtain ⟨he, hf, -, -⟩ := s2Frame_data h2 hq
    rcases hp' with rfl | rfl
    · show IsDvPure (s2Frame h2 hq) (wFactor (6, 1)) 6 1
      rw [wFactor_six]
      refine ⟨?_, ?_⟩
      · rw [gsix_eq, mem_dvSideSet_g h2 hq (by norm_num) Nat.one_pos]
        exact Or.inl ⟨rfl, by omega⟩
      · have hnd : (gsix : Polynomial O).natDegree
            / ((s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁) = 1 := by
          rw [natDegree_gsix, he, hf]
        rw [hnd, gsix_eq, mem_dvSideSet_g h2 hq (by norm_num) Nat.one_pos]
        exact Or.inr ⟨rfl, by omega⟩
    · show IsDvPure (s2Frame h2 hq) (wFactor (4, 1)) 4 1
      rw [wFactor_four]
      refine ⟨?_, ?_⟩
      · rw [gfour_eq, mem_dvSideSet_g h2 hq (by norm_num) Nat.one_pos]
        exact Or.inl ⟨rfl, by omega⟩
      · have hnd : (gfour : Polynomial O).natDegree
            / ((s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁) = 1 := by
          rw [natDegree_gfour, he, hf]
        rw [hnd, gfour_eq, mem_dvSideSet_g h2 hq (by norm_num) Nat.one_pos]

  hdeg := by
    intro p hp hne
    have hp' : p = ((6 : ℕ), (1 : ℕ)) ∨ p = ((4 : ℕ), (1 : ℕ)) := by
      simpa [wSlopes, Prod.ext_iff] using hp
    obtain ⟨he, hf, -, -⟩ := s2Frame_data h2 hq
    rcases hp' with rfl | rfl
    · show (wFactor (6, 1) : Polynomial O).natDegree
          = ((s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁)
            * (1 * dvSideDeg (s2Frame h2 hq) (fw : Polynomial O) 6 1 hne)
      rw [wFactor_six, natDegree_gsix, he, hf, dvSideDeg_fw_six h2 hq hne]
    · show (wFactor (4, 1) : Polynomial O).natDegree
          = ((s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁)
            * (1 * dvSideDeg (s2Frame h2 hq) (fw : Polynomial O) 4 1 hne)
      rw [wFactor_four, natDegree_gfour, he, hf, dvSideDeg_fw_four h2 hq hne]
  hbelow_monic := monic_one
  hbelow := by
    intro u ℓ hl hcop hfloor hne
    have hzero : ∀ x ∈ dvSideSet (s2Frame h2 hq) (1 : Polynomial O) u ℓ, x = 0 := by

      have hb := (mem_dvSideSet_iff.mp hx).1

    show (dvSideMax (s2Frame h2 hq) (1 : Polynomial O) u ℓ hne
        - dvSideMin (s2Frame h2 hq) (1 : Polynomial O) u ℓ hne) / ℓ = 0
    have hmax : dvSideMax (s2Frame h2 hq) (1 : Polynomial O) u ℓ hne = 0 :=
      hzero _ (Finset.max'_mem _ hne)

    simp
  hprod := by

    show (fw : Polynomial O) = ∏ p ∈ insert ((6 : ℕ), (1 : ℕ)) {((4 : ℕ), (1 : ℕ))}, wFactor p
    rw [Finset.prod_insert (by decide), Finset.prod_singleton, wFactor_six, wFactor_four]
    rfl
  hsides := by
    intro u ℓ hl hcop hfloor
    constructor
    · intro hm
      have hm' : (u = 6 ∧ ℓ = 1) ∨ (u = 4 ∧ ℓ = 1) := by
        simpa [wSlopes, Prod.ext_iff] using hm
      rcases hm' with ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩
      · have h0 : 0 ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 6 1 := by
          rw [mem_dvSideSet_fw h2 hq Nat.one_pos]
          exact Or.inl ⟨rfl, by omega⟩
        exact ⟨⟨0, h0⟩, by rw [dvSideDeg_fw_six h2 hq ⟨0, h0⟩]; norm_num⟩
      · have h1 : 1 ∈ dvSideSet (s2Frame h2 hq) (fw : Polynomial O) 4 1 := by

          exact Or.inr (Or.inl ⟨rfl, by omega⟩)
        exact ⟨⟨1, h1⟩, by rw [dvSideDeg_fw_four h2 hq ⟨1, h1⟩]; norm_num⟩
    · rintro ⟨hne, hdegpos⟩
      have hdd : dvSideDeg (s2Frame h2 hq) (fw : Polynomial O) u ℓ hne
          = ((dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ).max' hne
              - (dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ).min' hne) / ℓ := rfl
      have hgap : ℓ ≤ (dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ).max' hne
          - (dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ).min' hne := by
        by_contra hc
        rw [hdd, Nat.div_eq_of_lt (Nat.lt_of_not_le hc)] at hdegpos
        exact absurd hdegpos (lt_irrefl 0)
      have hminmem :=
        Finset.min'_mem (dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ) hne
      have hmaxmem :=
        Finset.max'_mem (dvSideSet (s2Frame h2 hq) (fw : Polynomial O) u ℓ) hne
      rw [mem_dvSideSet_fw h2 hq hl] at hminmem hmaxmem
      have hres : (u = 6 ∧ ℓ = 1) ∨ (u = 4 ∧ ℓ = 1) := by
        rcases hminmem with ⟨e1, e2⟩ | ⟨e1, e2⟩ | ⟨e1, e2⟩ <;>
          rcases hmaxmem with ⟨f1, f2⟩ | ⟨f1, f2⟩ | ⟨f1, f2⟩ <;>
          omega
      rcases hres with ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩ <;> simp [wSlopes]
