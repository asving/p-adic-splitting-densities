/-! ## 1. The two grammars, transcribed verbatim -/

/-- the §5 descent state: the current key and the current block factor (A-C.1, unchanged). -/
structure DescentState (O : Type) [CommRing O] where
  key : Polynomial O
  block : Polynomial O

/-- **THE A-C.1 DRAFT** (`leanspec/Leanspec/ChapC.lean`, struck): one step of §5's descent
grammar as first transcribed. Kept here only so that `draft_has_self_loop` can be stated. -/
inductive DescentStep (π : O) : DescentState O → DescentState O → Prop
  | jump : ∀ {F : KeyFrame O π} {H₀ : ℕ} {hpin : F.Pin H₀} (L : LevelDatum F H₀ hpin)
      (s s' : DescentState O),
      s.key = F.key → 2 ≤ L.ℓ * L.r.natDegree →
      HasLabel L s'.block → s'.block ∣ s.block →
      IsTestKey L s'.key → L.keyDeg₂ * 2 ≤ s'.block.natDegree →
      DescentStep π s s'
  | refine : ∀ (s s' : DescentState O) (u ℓ u' ℓ' : ℕ),
      0 < ℓ → 0 < ℓ' → s'.block = s.block →
      s'.key.natDegree = s.key.natDegree →
      (s.key - s'.key).natDegree < s.key.natDegree →
      (sideSet s.key s.block u ℓ).Nonempty →
      (sideSet s'.key s.block u' ℓ').Nonempty →
      u * ℓ' < u' * ℓ →
      DescentStep π s s'

/-- **THE A-C.6 REDRAFT**, binder-for-binder identical to `leanspec`'s `DescentStepR`. -/
inductive DescentStepR (π : O) : DescentState O → DescentState O → Prop
  | jump : ∀ {F : KeyFrame O π} {H₀ : ℕ} {hpin : F.Pin H₀} (L : LevelDatum F H₀ hpin)
      (s s' : DescentState O),
      s.key = F.key → 2 ≤ L.ℓ * L.r.natDegree →
      HasLabel L s'.block → s'.block ∣ s.block →
      IsTestKey L s'.key → L.keyDeg₂ * 2 ≤ s'.block.natDegree →
      DescentStepR π s s'
  | refine : ∀ (s s' : DescentState O) (u ℓ u' ℓ' : ℕ)
      (hne : (sideSet s.key s.block u ℓ).Nonempty)
      (hne' : (sideSet s'.key s.block u' ℓ').Nonempty),
      s.key.Monic → s'.key.Monic → 0 < s.key.natDegree →
      s'.block = s.block →
      s'.key.natDegree = s.key.natDegree →
      s'.key ≠ s.key →
      (s.key - s'.key).natDegree < s.key.natDegree →
      0 < u → 0 < ℓ → Nat.Coprime u ℓ →
      0 < u' → 0 < ℓ' → Nat.Coprime u' ℓ' →
      1 < (sideSet s.key s.block u ℓ).card →
      1 < (sideSet s'.key s.block u' ℓ').card →
      IsPure s.key s.block u ℓ →
      IsPure s'.key s.block u' ℓ' →
      2 ≤ sideDeg s.key s.block u ℓ hne →
      (∀ H₀ : ℕ, npHgt s.key s.block (sideMin s.key s.block u ℓ hne) = (H₀ : ℕ∞) →
        ∃ ζ : resField s.key, ζ ≠ 0 ∧
          resPoly π s.key s.block u ℓ hne H₀
            = (Polynomial.X - Polynomial.C ζ) ^ sideDeg s.key s.block u ℓ hne) →
      u * ℓ' < u' * ℓ →
      DescentStepR π s s'

/-! ## 2. LEG A — the separation: the draft self-loops, the redraft cannot -/

/-- **THE DRAFT'S DEFECT, re-proved here** (`C94_REFUTATION.descentStep_self`): every state of
positive key degree is a `DescentStep`-self-loop, which is the infinite chain. -/
theorem draft_has_self_loop (π : O) (s : DescentState O) (h : 0 < s.key.natDegree) :
    DescentStep π s s := by
  refine DescentStep.refine s s 0 1 1 1 Nat.one_pos Nat.one_pos rfl rfl ?_
    (sideSet_nonempty_gen _ _ _ _) (sideSet_nonempty_gen _ _ _ _) (by omega)
  simpa using h

/-- **LEG A(iii) — THE REDRAFT HAS NO SELF-LOOP, at either constructor.** `refine` is excluded by
the new `s'.key ≠ s.key`; `jump` by the degree arithmetic of `IsTestKey` (`keyDeg₂ = D′·ℓ·deg r`
with `2 ≤ ℓ·deg r` and `D′ = e₁f₁ > 0`). -/
theorem redraft_no_self_loop (π : O) (s : DescentState O) : ¬ DescentStepR π s s := by
  intro h
  cases h with
  | @jump F H₀ hpin L s s' hkey hjump _hlab _hdvd htest _hdeg =>
      have hdeg1 : s.key.natDegree = L.keyDeg₂ := htest.2.1
      rw [hkey, F.hdeg, LevelDatum.keyDeg₂] at hdeg1
      have hD : 0 < F.e₁ * F.f₁ := Nat.mul_pos F.he₁ F.hf₁
      have h2 : (F.e₁ * F.f₁) * 2 ≤ (F.e₁ * F.f₁) * (L.ℓ * L.r.natDegree) :=
        Nat.mul_le_mul_left _ hjump
      rw [← mul_assoc, ← hdeg1] at h2
      omega
  | refine s s' u ℓ u' ℓ' hne hne' _ _ _ _ _ hkeyne _ _ _ _ _ _ _ _ _ _ _ _ _ _ =>
      exact hkeyne rfl

/-- **LEG A(iii), the corollary the refutation needed.** The constant history — the refutation's
infinite chain — is not a `DescentStepR`-chain. -/
theorem redraft_constant_history_not_a_chain (π : O) (s : DescentState O) :
    ¬ (∀ n : ℕ, DescentStepR π ((fun _ => s) n) ((fun _ => s) (n + 1))) :=
  fun h => redraft_no_self_loop π s (h 0)

/-! ### 3.4 the residual polynomial of the side at the key `X`: `R = (Y − 1)²` -/

/-- the unscaled model `G` with `f = G.scaleRoots 2`: `G = x² − 6x + 1`, whose reduction mod `2`
is `Y² + 1 = (Y − 1)²`. -/
noncomputable def Gw (O : Type) [CommRing O] [IsDomain O] [IsDiscreteValuationRing O] :
    Polynomial O := X ^ 2 - C 6 * X + C 1

theorem Gw_natDegree : (Gw O).natDegree = 2 := by unfold Gw; compute_degree!

theorem Gw_coeff0 : (Gw O).coeff 0 = 1 := by simp [Gw]

theorem Gw_coeff1 : (Gw O).coeff 1 = -6 := by simp [Gw, coeff_one]

theorem Gw_coeff2 : (Gw O).coeff 2 = 1 := by simp [Gw, coeff_one]

theorem bl_eq_scale : (bl O) = (Gw O).scaleRoots ((2 : O) ^ 1) := by
  ext i
  rw [coeff_scaleRoots, Gw_natDegree]
  rcases i with _ | _ | _ | n
  · rw [bl_coeff0, Gw_coeff0]; norm_num
  · rw [bl_coeff1, Gw_coeff1]; norm_num
  · rw [bl_coeff2, Gw_coeff2]; norm_num
  · rw [coeff_eq_zero_of_natDegree_lt (by rw [bl_natDegree]; omega),
      coeff_eq_zero_of_natDegree_lt (by rw [Gw_natDegree]; omega), zero_mul]

theorem isKey_X' : IsKey (X : Polynomial O) where
  monic := monic_X
  pos := by simp
  irred := by rw [Polynomial.map_X]; exact irreducible_X

theorem two_eq_zero_resField : (2 : resField (X : Polynomial O)) = 0 := by
  have h2r : (2 : ResidueField O) = 0 := by
    rw [← map_ofNat (IsLocalRing.residue O) 2]
    exact (IsLocalRing.residue_eq_zero_iff _).2 ((IsLocalRing.mem_maximalIdeal _).2 h2.not_isUnit)
  have hof := congrArg (AdjoinRoot.of ((X : Polynomial O).map (IsLocalRing.residue O))) h2r
  rw [map_ofNat, map_zero] at hof
  exact hof

theorem six_residue_zero : IsLocalRing.residue O (6 : O) = 0 := by
  refine (IsLocalRing.residue_eq_zero_iff _).2 ?_
  have h2m : (2 : O) ∈ IsLocalRing.maximalIdeal O :=
    (IsLocalRing.mem_maximalIdeal _).2 h2.not_isUnit
  have := Ideal.mul_mem_left (IsLocalRing.maximalIdeal O) 3 h2m
  norm_num at this
  exact this

include h2 in
/-- **the residual read of the witness side**: `R_{1/1}(f) = (Y − 1)²` over `resField X`. The
route is B.59's `resPoly_X_eq_map_of_scaleRoots` (`f = G.scaleRoots 2`), then `G mod 2 = Y² + 1`,
then `Y² + 1 = (Y − 1)²` in residue characteristic `2`. -/
theorem resPoly_bl (H₀ : ℕ) (hH : npHgt X (bl O) 0 = (H₀ : ℕ∞)) :
    resPoly (2 : O) X (bl O) 1 1 (hne_bl h2) H₀
      = (X - C (1 : resField (X : Polynomial O))) ^ 2 := by
  rw [resPoly_X_eq_map_of_scaleRoots h2 bl_monic (by rw [Gw_natDegree, bl_natDegree])
    bl_eq_scale (pure_bl h2) (hne_bl h2) hH]
  have hmap : (Gw O).map (IsLocalRing.residue O) = X ^ 2 + 1 := by
    simp only [Gw, Polynomial.map_add, Polynomial.map_sub, Polynomial.map_mul, Polynomial.map_pow,
      Polynomial.map_X, Polynomial.map_C, Polynomial.map_one, six_residue_zero h2, map_one]
    simp
  rw [hmap]
  have h2poly : (2 : Polynomial (resField (X : Polynomial O))) = 0 := by
    rw [← map_ofNat (Polynomial.C (R := resField (X : Polynomial O))) 2,
      two_eq_zero_resField h2, map_zero]
  simp only [Polynomial.map_add, Polynomial.map_pow, Polynomial.map_X, Polynomial.map_one]
  rw [map_one]
  ring_nf
  rw [h2poly]
  ring

/-! ### 3.5 the witness step itself -/

include h2 in
/-- **LEG B — THE REDRAFTED `refine` CONSTRUCTOR IS INHABITED.** Block `f = x² − 12x + 4`,
key `x → x − 2`, slope `1/1 → 2/1`: every one of the twenty-two clause slots holds. -/
theorem witness_step :
    DescentStepR (2 : O) ⟨X, bl O⟩ ⟨ky O, bl O⟩ := by
  letI : Field (resField (X : Polynomial O)) := instFieldResField isKey_X'
  refine DescentStepR.refine ⟨X, bl O⟩ ⟨ky O, bl O⟩ 1 1 2 1 (hne_bl h2) (hne_ky h2)
    monic_X ky_monic (by simp) rfl (by rw [ky_natDegree, natDegree_X]) ?_ ?_
    Nat.one_pos Nat.one_pos (Nat.coprime_one_right 1)
    (by norm_num) Nat.one_pos (Nat.coprime_one_right 2)
    (sideSet_bl_card h2) (sideSet_ky_card h2) (pure_bl h2) (pure_ky h2) ?_ ?_ (by norm_num)
  · -- the keys differ: `x − 2 ≠ x`
    intro hEq
    have h := congrArg (fun p : Polynomial O => X - p) hEq
    simp only [ky, sub_sub_cancel, sub_self] at h
    exact h2.ne_zero ((Polynomial.C_eq_zero).1 h)
  · -- the recentring has strictly lower degree
    show ((X : Polynomial O) - ky O).natDegree < (X : Polynomial O).natDegree
    rw [show (X : Polynomial O) - ky O = C 2 from by rw [ky]; ring, natDegree_C, natDegree_X]
    norm_num
  · -- the side has residual degree `2`
    rw [sideDeg_bl h2]
  · -- the residual is a SQUARE of a nonzero linear form
    intro H₀ hH
    refine ⟨1, one_ne_zero, ?_⟩
    rw [sideDeg_bl h2]
    exact resPoly_bl h2 H₀ (by rw [← sideMin_bl h2]; exact hH)
