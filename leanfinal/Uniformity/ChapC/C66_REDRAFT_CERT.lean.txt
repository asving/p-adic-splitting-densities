/-! ## 3. LEG B — the satisfying witness `gw = X² + 2` at `(u, ℓ) = (4, 1)` -/

noncomputable def gw (O : Type) [CommRing O] [IsDomain O] [IsDiscreteValuationRing O] :

theorem gw_monic : (gw O).Monic := by
  have : (gw O) = Polynomial.X ^ 2 + Polynomial.C 2 := rfl
  rw [this]
  exact (Polynomial.monic_X_pow 2).add_of_left (by
    refine lt_of_le_of_lt Polynomial.degree_C_le ?_
    rw [Polynomial.degree_X_pow]
    decide)

theorem gw_natDegree : (gw O).natDegree = 2 := by
  have : (gw O).degree = 2 := by
    rw [show (gw O) = Polynomial.X ^ 2 + Polynomial.C 2 from rfl]
    rw [Polynomial.degree_add_eq_left_of_degree_lt (by
      refine lt_of_le_of_lt Polynomial.degree_C_le ?_
      rw [Polynomial.degree_X_pow]; decide)]
    exact Polynomial.degree_X_pow 2
  exact Polynomial.natDegree_eq_of_degree_eq_some this

theorem gw_divmod :
    (gw O) /ₘ (s2Key O) = 1 ∧ (gw O) %ₘ (s2Key O) = Polynomial.C ((2 : O) ^ 2) := by
  refine Polynomial.div_modByMonic_unique 1 _ s2Key_monic ⟨?_, ?_⟩
  · simp only [gw, s2Key, show ((2 : O) ^ 2) = 4 from by norm_num, map_ofNat]
    ring
  · rw [s2Key_deg]
    exact lt_of_le_of_lt Polynomial.degree_C_le (by decide)

  · rw [dvside_gw h2 hq]; simp
  · rw [show (gw O).natDegree / ((s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁) = 1 from by
        rw [gw_natDegree, e1_eq, f1_eq]]
    rw [dvside_gw h2 hq]; simp

theorem dvsidemin_gw : dvSideMin (s2Frame h2 hq) (gw O) 4 1 (hne_gw h2 hq) = 0 :=
  le_antisymm (Finset.min'_le _ 0 (by rw [dvside_gw h2 hq]; simp)) (Nat.zero_le _)

theorem dvsidemin_gwsq :
    dvSideMin (s2Frame h2 hq) ((gw O) * (gw O)) 4 1 (hne_gwsq h2 hq) = 0 :=
  le_antisymm (Finset.min'_le _ 0 (by rw [dvside_gwsq h2 hq]; simp)) (Nat.zero_le _)

theorem hp_gw : dvHgt (s2Frame h2 hq) (gw O)
    (dvSideMin (s2Frame h2 hq) (gw O) 4 1 (hne_gw h2 hq)) = ((4 : ℕ) : ℕ∞) := by
  rw [dvsidemin_gw h2 hq, dvhgt_gw_0 h2 hq]; rfl

theorem hp_gwsq : dvHgt (s2Frame h2 hq) ((gw O) * (gw O))
    (dvSideMin (s2Frame h2 hq) ((gw O) * (gw O)) 4 1 (hne_gwsq h2 hq)) = ((8 : ℕ) : ℕ∞) := by
  rw [dvsidemin_gwsq h2 hq, dvhgt_gwsq_0 h2 hq]; rfl

/-! ### the five A-C.6 binders, at the witness -/

theorem gw_hfloor : 1 * ((s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁ * (s2Frame h2 hq).h) < 4 := by
  rw [e1_eq h2 hq, f1_eq h2 hq, h_eq h2 hq]; omega

theorem gw_hdg : (s2Frame h2 hq).e₁ * (s2Frame h2 hq).f₁ ∣ (gw O).natDegree := by
  rw [e1_eq h2 hq, f1_eq h2 hq, gw_natDegree]

theorem gw_hpos : 0 < (gw O).natDegree := by rw [gw_natDegree]; norm_num

/-- **LEG B — SATISFIABILITY.** Every binder of the A-C.6 redraft is discharged at the LANDED
`s2Frame`, so the redraft delivers its conclusion there: the fences that kill the refuting pair
do NOT empty the statement. -/
theorem redraft_applies_at_s2Frame (hax : C66StatementR) :
    ∃ c : (s2Frame h2 hq).stageField 1 (s2Frame_pin h2 hq), c ≠ 0 ∧
      dvResPoly (s2Frame h2 hq) 1 (s2Frame_pin h2 hq) ((gw O) * (gw O)) 4 1
          (hne_gwsq h2 hq) 8 (hp_gwsq h2 hq)
        = Polynomial.C c *
            (dvResPoly (s2Frame h2 hq) 1 (s2Frame_pin h2 hq) (gw O) 4 1
                (hne_gw h2 hq) 4 (hp_gw h2 hq)
              * dvResPoly (s2Frame h2 hq) 1 (s2Frame_pin h2 hq) (gw O) 4 1
                  (hne_gw h2 hq) 4 (hp_gw h2 hq)) :=
  hax O 2 (s2Frame h2 hq) h2 1 (s2Frame_pin h2 hq) (gw O) (gw O)
    (gw_monic) (gw_monic) 4 1 Nat.one_pos (Nat.coprime_one_right 4)
    (gw_hfloor h2 hq) (gw_hdg h2 hq) (gw_hdg h2 hq) (gw_hpos) (gw_hpos)
    (pure_gw h2 hq) (pure_gw h2 hq) (hne_gw h2 hq) (hne_gw h2 hq) (hne_gwsq h2 hq)
    4 4 8 (hp_gw h2 hq) (hp_gw h2 hq) (hp_gwsq h2 hq)

/-! ## 4. LEG C — the degree bookkeeping at the witness is CONSISTENT (`2 = 1 + 1`) -/

theorem dvsidedeg_gw : dvSideDeg (s2Frame h2 hq) (gw O) 4 1 (hne_gw h2 hq) = 1 := by
  have hmax : dvSideMax (s2Frame h2 hq) (gw O) 4 1 (hne_gw h2 hq) = 1 := by
    refine le_antisymm (Finset.max'_le _ _ _ ?_)
      (Finset.le_max' _ 1 (by rw [dvside_gw h2 hq]; simp))
    intro y hy
    rw [dvside_gw h2 hq] at hy
    rcases Finset.mem_insert.mp hy with rfl | hy
    · norm_num
    · exact le_of_eq (by simpa using hy)
  rw [dvSideDeg, hmax, dvsidemin_gw h2 hq]

theorem dvsidedeg_gwsq :
    dvSideDeg (s2Frame h2 hq) ((gw O) * (gw O)) 4 1 (hne_gwsq h2 hq) = 2 := by
  have hmax : dvSideMax (s2Frame h2 hq) ((gw O) * (gw O)) 4 1 (hne_gwsq h2 hq) = 2 := by
    refine le_antisymm (Finset.max'_le _ _ _ ?_)
      (Finset.le_max' _ 2 (by rw [dvside_gwsq h2 hq]; simp))
    intro y hy
    rw [dvside_gwsq h2 hq] at hy
    rcases Finset.mem_insert.mp hy with rfl | hy
    · norm_num
    · exact le_of_eq (by simpa using hy)
  rw [dvSideDeg, hmax, dvsidemin_gwsq h2 hq]

/-- **LEG C.** The witness's residual degrees ADD, as the conclusion `R(g·g′) = C c · R(g)R(g′)`
requires of a nonzero scalar `c`: `2 = 1 + 1`. (At the refuting pair the same bookkeeping read
`1 = 0 + 0` — `C66_REFUTATION`'s `dvsidedeg_X`/`dvsidedeg_XX` — which is the refutation.) -/
theorem degree_bookkeeping_consistent :
    dvSideDeg (s2Frame h2 hq) ((gw O) * (gw O)) 4 1 (hne_gwsq h2 hq)
      = dvSideDeg (s2Frame h2 hq) (gw O) 4 1 (hne_gw h2 hq)
        + dvSideDeg (s2Frame h2 hq) (gw O) 4 1 (hne_gw h2 hq) := by
  rw [dvsidedeg_gw h2 hq, dvsidedeg_gwsq h2 hq]
